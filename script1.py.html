#functions assignment
#Circle function

def calculate_circle_area(radius):
    pi = 3.14159
    area = pi * (radius ** 2)
    return f"{area:.2f}"

radius_input = float(input("Enter radius of circle: "))
area_result = calculate_circle_area(radius_input)
print(f"Area of circle: {area_result}")

#Taxes function
def calculate_total_due(money, tax):
    total_due = money + (money * tax)
    return total_due

money_input = float(input("Enter money amount: "))
tax_input = float(input("Enter tax rate (as a decimal, e.g., 0.07): "))
total_result = calculate_total_due(money_input, tax_input)
print(f"Total due: ${total_result:.2f}")

#temperature function
def fahrenheit_to_celsius(fahrenheit):
    celsius = (fahrenheit - 32) * (5 / 9)
    return celsius

fahrenheit_input = float(input("Enter temperature in Fahrenheit: "))
celsius_result = fahrenheit_to_celsius(fahrenheit_input)
print(f"Temperature in Celsius: {celsius_result:.2f}")
